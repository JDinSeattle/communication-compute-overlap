cmake_minimum_required(VERSION 3.24)
project(communication_compute_overlap LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
find_package(CUDAToolkit REQUIRED)
set(MSCCLPP_ROOT "" CACHE PATH "Pinned MSCCL++ installation prefix")
find_path(MSCCLPP_INCLUDE mscclpp/core.hpp HINTS "${MSCCLPP_ROOT}/include" REQUIRED)
find_library(MSCCLPP_LIBRARY mscclpp HINTS "${MSCCLPP_ROOT}/lib" REQUIRED)
add_executable(overlap src/overlap.cu)
target_include_directories(overlap PRIVATE "${MSCCLPP_INCLUDE}")
target_link_libraries(overlap PRIVATE "${MSCCLPP_LIBRARY}" CUDA::cudart CUDA::cuda_driver)
target_compile_options(overlap PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
set_target_properties(overlap PROPERTIES BUILD_RPATH "${MSCCLPP_ROOT}/lib")
